cmake_minimum_required(VERSION 3.16)
project(libabhed VERSION 2.1.1 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
# installed binaries find liboqs and libabhed without LD_LIBRARY_PATH
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH ON)
set(CMAKE_INSTALL_RPATH "${CMAKE_INSTALL_PREFIX}/lib")
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
set(ABHED_ALG_MTL 244 CACHE STRING "DNSSEC algorithm number for ML-DSA-44-MTL-SHAKE-128")
set(ABHED_OPT_SIGTAG 65001 CACHE STRING "EDNS(0) option code for SigTag")
set(ABHED_OPT_ABHED 65002 CACHE STRING "EDNS(0) option code for the ABHED profile flags")
option(ABHED_SANITIZE "Build with AddressSanitizer and UBSan" OFF)

find_package(PkgConfig)
find_package(Threads REQUIRED)
find_path(OQS_INCLUDE oqs/oqs.h HINTS $ENV{OQS_ROOT}/include /usr/local/include)
find_library(OQS_LIB oqs HINTS $ENV{OQS_ROOT}/lib /usr/local/lib)
if(NOT OQS_INCLUDE OR NOT OQS_LIB)
  message(FATAL_ERROR "liboqs not found. Build it first (deploy/build.sh does) or set OQS_ROOT.")
endif()

add_library(abhed SHARED src/keccak.c src/mtl.c src/mldsa.c src/rrsig.c src/lcache.c src/nodeset.c)
add_library(abhed_static STATIC src/keccak.c src/mtl.c src/mldsa.c src/rrsig.c src/lcache.c src/nodeset.c)
set_target_properties(abhed_static PROPERTIES OUTPUT_NAME abhed POSITION_INDEPENDENT_CODE ON)
foreach(t abhed abhed_static)
  target_include_directories(${t} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${OQS_INCLUDE})
  target_compile_definitions(${t} PUBLIC ABHED_ALG_MTL=${ABHED_ALG_MTL} ABHED_OPT_SIGTAG=${ABHED_OPT_SIGTAG} ABHED_OPT_ABHED=${ABHED_OPT_ABHED})
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wshadow -Wformat=2 -fstack-protector-strong -D_FORTIFY_SOURCE=2)
  target_link_libraries(${t} PUBLIC ${OQS_LIB} Threads::Threads)
  if(ABHED_SANITIZE)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
set_target_properties(abhed PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 2)

enable_testing()
add_executable(test_libabhed tests/test_libabhed.c)
target_link_libraries(test_libabhed abhed_static)
add_test(NAME libabhed COMMAND test_libabhed ${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors.txt)

include(GNUInstallDirs)
install(TARGETS abhed abhed_static LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/abhed/abhed.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abhed)
configure_file(libabhed.pc.in libabhed.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/libabhed.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
